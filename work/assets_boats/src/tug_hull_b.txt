  const L = 11, ZS = -5.5, ST = 64, B2 = 2.3, TOP = 3.0, DECK = 2.55;
  const half = (t) => { const u = 2 * t - 1, p = u < 0 ? 3.2 : 2.1; return B2 * Math.pow(Math.max(0, 1 - Math.pow(Math.abs(u), p)), 1 / p); };
  const sheer0 = (t) => (t < 0.45 ? 2.55 + 0.15 * Math.pow((0.45 - t) / 0.45, 2) : 2.55 + 0.75 * Math.pow((t - 0.45) / 0.55, 2)) + 0.45;
  const bottom = (t) => t < 0.25 ? 0.2 + 1.25 * Math.pow((0.25 - t) / 0.25, 1.8) : t > 0.7 ? 0.2 + (2.9 - 0.35) * Math.pow((t - 0.7) / 0.3, 1.6) : 0.2;
  const N = 0.42, MM = 1.3;
  const zOf = (t) => ZS + t * L, tAt = (z) => Math.min(1, Math.max(0, (z - ZS) / L));
  const aOf = (t, y) => { const ys = sheer0(t), yb = bottom(t); const q = Math.min(1, Math.max(0, (ys - y) / Math.max(1e-4, ys - yb))); return Math.acos(Math.pow(q, 1 / MM)); };
  const xAt = (t, y) => half(t) * Math.pow(Math.sin(aOf(t, y)), N);
  const sheer = () => TOP;
  const outline = (y, k) => { const R = []; for (let i = 0; i <= ST; i++) { const t = i / ST; if (bottom(t) >= y - 0.005) continue; R.push([Math.max(0.01, xAt(t, y) * (k || 1)), zOf(t)]); } return R; };
  const shapeOf = (R, Cls) => { const sh = new (Cls || THREE.Shape)(); sh.moveTo(R[0][0], -R[0][1]); for (const [x, z] of R.slice(1)) sh.lineTo(x, -z); for (let k = R.length - 1; k >= 0; k--) sh.lineTo(-R[k][0], -R[k][1]); sh.closePath(); return sh; };
  const slab = (y0, y1, yo, mat, name, hole) => {
    const R = outline(yo); if (R.length < 2) return; const sh = shapeOf(R);
    if (hole) sh.holes.push(shapeOf(outline(yo, hole).filter((p, i, a) => i > 1 && i < a.length - 2), THREE.Path));
    const bv = Math.min(0.05, (y1 - y0) * 0.22);
    const geo = new THREE.ExtrudeGeometry(sh, { depth: y1 - y0 - 2 * bv, bevelEnabled: true, bevelThickness: bv, bevelSize: bv * 0.7, bevelSegments: 2, curveSegments: 4 });
    geo.rotateX(-Math.PI / 2); geo.translate(0, y0 + bv, 0); return add(geo, mat, name || 'hull');
  };
